#### 4.1. Estrutura do Formulário de Busca (Frontend)

 A página consulta.html envia os parâmetros de busca para o Django utilizando o método GET. Isso permite que os filtros fiquem visíveis na URL do navegador (útil para compartilhamento de links ou recarregamento de página).
 
 Os principais campos mapeados no formulário são:
 * Patrimônio (Atual/Antigo): Filtro direto por texto exato.

 * Equipamento (Modelo): Menu do tipo select alimentado dinamicamente pelas marcas/modelos.
 * Responsável Unidade: Menu do tipo select para filtrar por responsáveis cadastrados.

    filtro_responsaveis = request.GET.get('responsavel', '').strip()

    lista_equipamentos = Patrimonio.objects.select_related('tipo_hardware', 'modelo__marca') \

            'modelo__marca__nome', 
            'modelo__nome'

            'unidade_adm', 
            'responsavel',

            query = query.filter(modelo_id=filtro_equipamento)

            query = query.filter(responsavel_id=filtro_responsaveis)

def gerar_pdf_relatorio(dados_banco, titulo_dinamico, marca_modelo_dinamico, nome_arquivo="relatorio.pdf"):
    """Gera o arquivo PDF baseado no modelo visual fornecido"""
    doc = SimpleDocTemplate(
        nome_arquivo, 
        pagesize=letter,
        rightMargin=0.5*inch, 
        leftMargin=0.5*inch,
        topMargin=0.5*inch, 
        bottomMargin=0.5*inch
    )
    
    story = []
    styles = getSampleStyleSheet()
    
    # Configurações de Estilos de Texto
    style_titulo = ParagraphStyle(
        'TituloRelatorio',
        parent=styles['Heading1'],
        fontName='Helvetica-Bold',
        fontSize=22,
        leading=26,
        textColor=colors.HexColor('#0f4c81'),
        alignment=1
    )
    
    style_cabecalho_texto = ParagraphStyle(
        'CabecalhoTexto',
        parent=styles['Normal'],
        fontName='Helvetica',
        fontSize=10,
        leading=14,
        textColor=colors.black
    )

        'TextoHeaderTabela',
        parent=styles['Normal'],
        fontName='Helvetica-Bold',
        fontSize=10,
        leading=12,
        textColor=colors.white
    )
    
    style_celula = ParagraphStyle(
        'TextoCelula',
        parent=styles['Normal'],
        fontName='Helvetica',
        fontSize=9,
        leading=11,
        textColor=colors.black
    )

    # 1. Título do Relatório
    story.append(Paragraph(titulo_dinamico, style_titulo))
    story.append(Spacer(1, 15))
    
    # 2. Informações Dinâmicas do Cabeçalho
    total_itens = len(dados_banco)
    data_atual = datetime.now().strftime("%d/%m/%Y às %H:%M")

    <b>Total de Itens Encontrados:</b> {total_itens}<br/>
    <b>Data de Emissão:</b> {data_atual}
    """
    story.append(Paragraph(info_html, style_cabecalho_texto))
    story.append(Spacer(1, 15))
    
    # 3. Estrutura de dados para a tabela (Inicia com o cabeçalho)
    table_data = [[
        Paragraph("Patrimônio", style_th),
        Paragraph("Unidade Administrativa", style_th),
        Paragraph("Responsável", style_th),
        Paragraph("Garantia", style_th)

        fontSize=8,
        textColor=colors.HexColor('#4A5568')

        ('VALIGN', (0, 0), (-1, -1), 'TOP'),

    for row in dados_banco:

        contagem_responsaveis[resp] = contagem_responsaveis.get(resp, 0) + 1

    responsavel_anterior = None

    garantia_anterior = None

    for idx, row in enumerate(dados_banco):

        unidade_atual = row['unidade_administrativa']
        garantia_atual = row['garantia']

        # 💡 NOVA LÓGICA DE CONTROLE DE OMISSÃO:
        if responsavel_atual == responsavel_anterior:
            exibir_responsavel = ""

            Paragraph(exibir_responsavel, style_celula),
            Paragraph(exibir_garantia, style_celula)

        responsavel_anterior = responsavel_atual
        unidade_anterior = unidade_atual  # <--- Salva a unidade atual para comparar na próxima volta

        qtd_total = contagem_responsaveis[responsavel_atual]
        eh_ultimo_do_responsavel = (idx == len(dados_banco) - 1) or (dados_banco[idx+1]['responsavel'] != responsavel_atual)
        
        # Se chegou ao fim do bloco daquela pessoa, insere o totalizador cinza

                linha_subtotal = len(table_data) - 1

    col_widths = [72, 216, 162, 90]
    t = Table(table_data, colWidths=col_widths, repeatRows=1)
